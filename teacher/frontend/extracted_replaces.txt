export default function GlobalSearchModal({
  isOpen,
  onClose,
  searchQuery,
  setSearchQuery,
  searchResults,
  onSelectStudent,
}) {
  const inputRef = useRef(null);
  const navigate = useNavigate();

  useEffect(() => {
    if (isOpen) {
      setTimeout(() => inputRef.current?.focus(), 50);
    }
  }, [isOpen]);

  useEffect(() => {
    const handleKeyDown = (e) => {
      if ((e.metaKey || e.ctrlKey) && e.key === 'k') {
        e.preventDefault();
      }
      if (e.key === 'Escape' && isOpen) {
        onClose();
      }
    };
    window.addEventListener('keydown', handleKeyDown);
    return () => window.removeEventListener('keydown', handleKeyDown);
  }, [isOpen, onClose]);

  if (!isOpen) return null;

  const handleResultClick = (res) => {
    onClose();
    if (res.type === 'student' && onSelectStudent) {
      onSelectStudent(res.data);
    } else if (res.type === 'class') {
      navigate('/classes');
    } else if (res.type === 'assignment') {
      navigate('/grading');
    } else if (res.type === 'concept') {
      navigate('/');
    }
  };

  return (
    <div className="fixed inset-0 z-50 bg-black/75 backdrop-blur-sm flex items-start justify-center pt-20 p-4 animate-in fade-in duration-150">
      <div
        className="w-full max-w-2xl bg-[#131B2E] border border-[#243354] rounded-2xl shadow-2xl overflow-hidden flex flex-col max-h-[80vh]"
